# SQL Check Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A test-only package, `internal/sqlcheck`, that finds every SQL string under `internal/` and checks it against the migrated schema. Each statement must prepare, its bind parameters must match the arguments passed, and its columns must match the Scan that reads them. SQL built at run time must be listed with a reason.

**Architecture:** There are four layers, each its own `_test.go` file, so nothing in the package ships in the binary:
- a tokenizer that splits statements and counts parameters;
- an AST scanner that finds the calls carrying SQL, including pass-through wrappers it discovers;
- a resolver that turns each SQL argument into the strings it can hold;
- the checks, run over the whole tree against a database copied from `storetest`.

**Tech Stack:** Go standard library (`go/ast`, `go/parser`, `go/token`, `database/sql`), `modernc.org/sqlite` (already a dependency), and `internal/store/storetest`.

**Spec:** `docs/superpowers/specs/2026-10-10-sql-check-design.md`

## Review Focus

1. **`LIKE ? ESCAPE '\'`** (the forms search). A backslash inside a SQL string is literal, so the string closes at the next `'`, and the statement binds 1 parameter. Pinned by the `like escape` case in Task 1.
2. **A query and its Scan inside a closure passed to a transaction helper** (`d.tx(ctx, func(tx *sql.Tx) error {…})`, used all over `store/sqlite`). The Scan is still matched to its query. Pinned by `TestScansInsideAClosure` in Task 2.
3. **A package variable reassigned anywhere in its package**, such as an `init` that changes it. It is dynamic, not resolved to its first value. Pinned in `TestDynamicSitesSayWhy` in Task 2.
4. **`LIMIT ?` on a read whose Scan is counted.** Binding NULL fails with "datatype mismatch", so the column count retries with 0 and reports no problem. Pinned by `limit param` in Task 3.
5. **A read that would write if run** (`INSERT … RETURNING`, or a `WITH … INSERT` reached through a read call). Counting its columns must leave the database unchanged. Pinned by `TestColumnsRollsBack` in Task 3.

package sqlcheck

import (
	"fmt"
	"regexp"
	"strconv"
	"strings"
)

// params is what one statement binds, numbered the way SQLite numbers
// them: ?NNN takes N, a bare ? or a new name takes one more than the
// highest number so far, and a name seen before keeps its number.
type params struct {
	max   int            // sqlite3_bind_parameter_count: arguments the statement needs
	names map[int]string // number → name without its prefix, for :x @x $x
}

// lex walks sql, skipping quoted text ('…' "…" `…` […]) and comments
// (-- and /* */), and calls visit for every other byte. visit returns the
// offset to continue from, so a caller can consume a token of several
// bytes. An unterminated quote or comment is an error.
func lex(sql string, visit func(i int) int) error {
	for i := 0; i < len(sql); {
		c := sql[i]
		switch {
		case c == '\'' || c == '"' || c == '`':
			j := i + 1
			for {
				k := strings.IndexByte(sql[j:], c)
				if k < 0 {
					return fmt.Errorf("unterminated %c at offset %d", c, i)
				}
				j += k + 1
				if j < len(sql) && sql[j] == c { // doubled quote: an escaped one
					j++
					continue
				}
				break
			}
			i = j
		case c == '[':
			k := strings.IndexByte(sql[i:], ']')
			if k < 0 {
				return fmt.Errorf("unterminated [ at offset %d", i)
			}
			i += k + 1
		case strings.HasPrefix(sql[i:], "--"):
			k := strings.IndexByte(sql[i:], '\n')
			if k < 0 {
				return nil
			}
			i += k + 1
		case strings.HasPrefix(sql[i:], "/*"):
			k := strings.Index(sql[i+2:], "*/")
			if k < 0 {
				return fmt.Errorf("unterminated /* at offset %d", i)
			}
			i += k + 4
		default:
			i = visit(i)
		}
	}
	return nil
}

// visible is sql with quoted text and comments blanked out.
func visible(sql string) (string, error) {
	var b strings.Builder
	err := lex(sql, func(i int) int { b.WriteByte(sql[i]); return i + 1 })
	return b.String(), err
}

var createTrigger = regexp.MustCompile(`(?i)\bCREATE\s+(TEMP\s+|TEMPORARY\s+)?TRIGGER\b`)

// statements splits sql on the semicolons outside quotes and comments and
// drops the empty pieces. A CREATE TRIGGER is refused: its body holds
// semicolons of its own, and no Go SQL string needs one.
func statements(sql string) ([]string, error) {
	var out []string
	start := 0
	add := func(s string) error {
		v, err := visible(s)
		if err != nil {
			return err
		}
		if strings.TrimSpace(v) == "" {
			return nil
		}
		if createTrigger.MatchString(v) {
			return fmt.Errorf("CREATE TRIGGER in a Go SQL string: its body holds ';', which this check cannot split")
		}
		out = append(out, strings.TrimSpace(s))
		return nil
	}
	var addErr error
	err := lex(sql, func(i int) int {
		if sql[i] == ';' && addErr == nil {
			addErr = add(sql[start:i])
			start = i + 1
		}
		return i + 1
	})
	if err != nil {
		return nil, err
	}
	if addErr != nil {
		return nil, addErr
	}
	if err := add(sql[start:]); err != nil {
		return nil, err
	}
	return out, nil
}

func isDigit(c byte) bool { return '0' <= c && c <= '9' }

func isIdent(c byte) bool {
	return c == '_' || c == '$' || isDigit(c) || 'a' <= c && c <= 'z' || 'A' <= c && c <= 'Z'
}

// bindings reports the parameters one statement binds.
func bindings(stmt string) (params, error) {
	p := params{names: map[int]string{}}
	byName := map[string]int{}
	err := lex(stmt, func(i int) int {
		c := stmt[i]
		if i > 0 && isIdent(stmt[i-1]) { // inside an identifier such as a$b
			return i + 1
		}
		switch c {
		case '?':
			j := i + 1
			for j < len(stmt) && isDigit(stmt[j]) {
				j++
			}
			if j == i+1 {
				p.max++
				return j
			}
			if n, _ := strconv.Atoi(stmt[i+1 : j]); n > p.max {
				p.max = n
			}
			return j
		case ':', '@', '$':
			j := i + 1
			for j < len(stmt) && isIdent(stmt[j]) && stmt[j] != '$' {
				j++
			}
			if j == i+1 {
				return j
			}
			name := stmt[i+1 : j]
			if _, ok := byName[name]; !ok {
				p.max++
				byName[name] = p.max
				p.names[p.max] = name
			}
			return j
		}
		return i + 1
	})
	return p, err
}

package sqlcheck

import (
	"fmt"
	"go/ast"
	"go/token"
	"strconv"
)

// maxCandidates caps the strings one argument resolves to, so a
// concatenation of switches cannot multiply without bound.
const maxCandidates = 64

// helpers stand in for calls that build a fragment of SQL: the
// placeholder list for an IN (…) prepares the same with one ?.
var helpers = map[string]string{"placeholders": "?"}

// resolver turns the SQL argument of one function's calls into the
// strings it can hold (D3).
type resolver struct {
	consts     map[string]ast.Expr
	types      map[string]*ast.StructType
	params     map[string]bool
	assigns    map[string][]ast.Expr // local name → every value assigned
	unresolved map[string]string     // local name → why its values cannot be known
	ranges     map[string]ast.Expr   // range value name → what is ranged over
	busy       map[string]bool       // names being resolved, to stop a cycle
}

func newResolver(fn *ast.FuncDecl, consts map[string]ast.Expr, types map[string]*ast.StructType) *resolver {
	r := &resolver{consts: consts, types: types, params: map[string]bool{},
		assigns: map[string][]ast.Expr{}, unresolved: map[string]string{},
		ranges: map[string]ast.Expr{}, busy: map[string]bool{}}
	names, _, _ := paramList(fn)
	for _, n := range names {
		r.params[n] = true
	}
	ast.Inspect(fn.Body, func(n ast.Node) bool {
		switch n := n.(type) {
		case *ast.AssignStmt:
			for i, l := range n.Lhs {
				id, ok := l.(*ast.Ident)
				if !ok || id.Name == "_" {
					continue
				}
				switch {
				case n.Tok != token.DEFINE && n.Tok != token.ASSIGN:
					r.unresolved[id.Name] = id.Name + " is changed with " + n.Tok.String()
				case len(n.Lhs) != len(n.Rhs):
					r.unresolved[id.Name] = id.Name + " is one of several results of a call"
				default:
					r.assigns[id.Name] = append(r.assigns[id.Name], n.Rhs[i])
				}
			}
		case *ast.ValueSpec:
			for i, id := range n.Names {
				if i < len(n.Values) {
					r.assigns[id.Name] = append(r.assigns[id.Name], n.Values[i])
				}
			}
		case *ast.RangeStmt:
			if id, ok := n.Value.(*ast.Ident); ok {
				r.ranges[id.Name] = n.X
			}
		}
		return true
	})
	return r
}

// resolve returns the strings e can hold, or why it cannot be known.
func (r *resolver) resolve(e ast.Expr) ([]string, string) {
	switch e := e.(type) {
	case *ast.BasicLit:
		if e.Kind == token.STRING {
			s, err := strconv.Unquote(e.Value)
			if err != nil {
				return nil, err.Error()
			}
			return []string{s}, ""
		}
	case *ast.ParenExpr:
		return r.resolve(e.X)
	case *ast.BinaryExpr:
		if e.Op != token.ADD {
			break
		}
		left, why := r.resolve(e.X)
		if why != "" {
			return nil, why
		}
		right, why := r.resolve(e.Y)
		if why != "" {
			return nil, why
		}
		if len(left)*len(right) > maxCandidates {
			return nil, "more than " + strconv.Itoa(maxCandidates) + " candidate strings"
		}
		var out []string
		for _, a := range left {
			for _, b := range right {
				out = append(out, a+b)
			}
		}
		return out, ""
	case *ast.Ident:
		if r.busy[e.Name] {
			return nil, e.Name + " is assigned from itself"
		}
		r.busy[e.Name] = true
		defer delete(r.busy, e.Name)
		if why, ok := r.unresolved[e.Name]; ok {
			return nil, why
		}
		if vals, ok := r.assigns[e.Name]; ok {
			return r.union(vals)
		}
		if over, ok := r.ranges[e.Name]; ok {
			lit, why := r.literal(over)
			if why != "" {
				return nil, why
			}
			return r.union(lit.Elts)
		}
		if r.params[e.Name] {
			return nil, e.Name + " is a parameter"
		}
		if v, ok := r.consts[e.Name]; ok {
			return r.resolve(v)
		}
		return nil, e.Name + " is not a constant, a local or an unchanged package variable"
	case *ast.CallExpr:
		name := calleeName(e)
		if s, ok := helpers[name]; ok {
			if _, isIdent := e.Fun.(*ast.Ident); isIdent {
				return []string{s}, ""
			}
		}
		return nil, "built by a call to " + exprString(e.Fun)
	case *ast.SelectorExpr:
		if x, ok := e.X.(*ast.Ident); ok {
			if over, ok := r.ranges[x.Name]; ok {
				return r.field(over, e.Sel.Name)
			}
		}
		return nil, exprString(e) + " is a field or another package's name"
	}
	return nil, fmt.Sprintf("%s (%T)", exprString(e), e)
}

// union resolves every expression and joins the results.
func (r *resolver) union(es []ast.Expr) ([]string, string) {
	var out []string
	for _, v := range es {
		got, why := r.resolve(v)
		if why != "" {
			return nil, why
		}
		out = append(out, got...)
	}
	if len(out) > maxCandidates {
		return nil, "more than " + strconv.Itoa(maxCandidates) + " candidate strings"
	}
	return out, ""
}

// field resolves name in every element of the slice literal over, which
// is a range loop's operand, written inline or held by a local.
func (r *resolver) field(over ast.Expr, name string) ([]string, string) {
	lit, why := r.literal(over)
	if why != "" {
		return nil, why
	}
	arr, ok := lit.Type.(*ast.ArrayType)
	if !ok {
		return nil, "ranges over a literal that is not a slice"
	}
	var st *ast.StructType
	switch t := arr.Elt.(type) {
	case *ast.StructType:
		st = t
	case *ast.Ident:
		st = r.types[t.Name]
	}
	if st == nil {
		return nil, "ranges over elements of an unknown struct type"
	}
	index := -1
	i := 0
	for _, f := range st.Fields.List {
		for _, n := range f.Names {
			if n.Name == name {
				index = i
			}
			i++
		}
	}
	if index < 0 {
		return nil, "no field " + name
	}
	var vals []ast.Expr
	for _, el := range lit.Elts {
		cl, ok := el.(*ast.CompositeLit)
		if !ok {
			return nil, "an element is not a struct literal"
		}
		var v ast.Expr
		for j, x := range cl.Elts {
			if kv, ok := x.(*ast.KeyValueExpr); ok {
				if k, ok := kv.Key.(*ast.Ident); ok && k.Name == name {
					v = kv.Value
				}
			} else if j == index {
				v = x
			}
		}
		if v == nil {
			return nil, "an element leaves " + name + " unset"
		}
		vals = append(vals, v)
	}
	return r.union(vals)
}

// literal is the slice literal a range loop runs over: written inline,
// held by a local assigned once, or by an unchanged package variable.
func (r *resolver) literal(over ast.Expr) (*ast.CompositeLit, string) {
	if id, ok := over.(*ast.Ident); ok {
		if vals := r.assigns[id.Name]; len(vals) == 1 {
			over = vals[0]
		} else if v, ok := r.consts[id.Name]; ok && vals == nil {
			over = v
		}
	}
	lit, ok := over.(*ast.CompositeLit)
	if !ok {
		return nil, "ranges over " + exprString(over) + ", not a slice literal"
	}
	return lit, ""
}

// exprString renders a short form of e for messages.
func exprString(e ast.Expr) string {
	switch e := e.(type) {
	case *ast.Ident:
		return e.Name
	case *ast.SelectorExpr:
		return exprString(e.X) + "." + e.Sel.Name
	case *ast.CallExpr:
		return exprString(e.Fun) + "(…)"
	case *ast.IndexExpr:
		return exprString(e.X) + "[…]"
	}
	return fmt.Sprintf("%T", e)
}

package sqlcheck

import (
	"fmt"
	"go/ast"
	"go/parser"
	"go/token"
	"io/fs"
	"os"
	"path/filepath"
	"sort"
	"strings"
)

// sink is a function or method that runs SQL: the argument at index sql
// is the statement, and the arguments from index args on are its
// parameters (args < 0: the parameters are not counted).
type sink struct{ sql, args int }

// baseSinks are the database/sql methods, matched by name on any receiver.
var baseSinks = map[string]sink{
	"ExecContext":     {sql: 1, args: 2},
	"QueryContext":    {sql: 1, args: 2},
	"QueryRowContext": {sql: 1, args: 2},
	"PrepareContext":  {sql: 1, args: -1},
}

// readers are the sinks whose rows a Scan reads (D6).
var readers = map[string]bool{"QueryContext": true, "QueryRowContext": true}

// stmtCalls are the methods of a prepared *sql.Stmt that bind its
// parameters: every argument after the context.
var stmtCalls = map[string]bool{"ExecContext": true, "QueryContext": true, "QueryRowContext": true}

// site is one call that carries SQL.
type site struct {
	pos   string   // file:line, the file relative to the scanned root
	key   string   // file:Func, the dynamic list's key
	sqls  []string // what the SQL argument resolves to; nil when dynamic
	why   string   // for a dynamic site, what did not resolve
	nargs []int    // argument counts passed for the parameters, one per call checked
	read  bool     // a database/sql read: its rows can be scanned
	scans []int    // Scan destination counts found for its rows
}

type file struct {
	rel string
	ast *ast.File
}

// scanSource parses every non-test .go file under root, keeping each
// directory's files together as a package, and returns their SQL sites.
func scanSource(root string) ([]site, error) {
	src := map[string]string{}
	err := filepath.WalkDir(root, func(path string, d fs.DirEntry, err error) error {
		if err != nil {
			return err
		}
		if d.IsDir() {
			if name := d.Name(); path != root && (name == "testdata" || name == "node_modules" || strings.HasPrefix(name, ".")) {
				return filepath.SkipDir
			}
			return nil
		}
		if !strings.HasSuffix(path, ".go") || strings.HasSuffix(path, "_test.go") {
			return nil
		}
		b, err := os.ReadFile(path)
		if err != nil {
			return err
		}
		rel, _ := filepath.Rel(root, path)
		src[filepath.ToSlash(rel)] = string(b)
		return nil
	})
	if err != nil {
		return nil, err
	}
	return scanFiles(src)
}

// scanFiles is scanSource over sources in memory, keyed by slash path:
// the files of one directory form one package.
func scanFiles(src map[string]string) ([]site, error) {
	fset := token.NewFileSet()
	pkgs := map[string][]file{}
	names := make([]string, 0, len(src))
	for name := range src {
		names = append(names, name)
	}
	sort.Strings(names)
	for _, name := range names {
		f, err := parser.ParseFile(fset, name, src[name], parser.SkipObjectResolution)
		if err != nil {
			return nil, err
		}
		dir := pathDir(name)
		pkgs[dir] = append(pkgs[dir], file{rel: name, ast: f})
	}

	sinks := map[string]sink{}
	for k, v := range baseSinks {
		sinks[k] = v
	}
	passThrough := discoverSinks(pkgs, sinks)

	var out []site
	dirs := make([]string, 0, len(pkgs))
	for dir := range pkgs {
		dirs = append(dirs, dir)
	}
	sort.Strings(dirs)
	for _, dir := range dirs {
		files := pkgs[dir]
		consts, types := packageDecls(files)
		for _, f := range files {
			for _, d := range f.ast.Decls {
				fn, ok := d.(*ast.FuncDecl)
				if !ok || fn.Body == nil {
					continue
				}
				out = append(out, funcSites(fset, f.rel, fn, sinks, passThrough[fn], consts, types)...)
			}
		}
	}
	return out, nil
}

func pathDir(p string) string {
	if i := strings.LastIndexByte(p, '/'); i >= 0 {
		return p[:i]
	}
	return "."
}

// calleeName is the name a call is made by: f(…) or x.f(…).
func calleeName(c *ast.CallExpr) string {
	switch f := c.Fun.(type) {
	case *ast.Ident:
		return f.Name
	case *ast.SelectorExpr:
		return f.Sel.Name
	}
	return ""
}

// paramList flattens a function's parameters to their names, in order,
// and reports which are strings and whether the last is variadic.
func paramList(fn *ast.FuncDecl) (names []string, isString []bool, variadic bool) {
	for _, f := range fn.Type.Params.List {
		id, str := f.Type.(*ast.Ident)
		_, variadic = f.Type.(*ast.Ellipsis)
		for _, n := range f.Names {
			names = append(names, n.Name)
			isString = append(isString, str && id.Name == "string")
		}
	}
	return names, isString, variadic
}

// discoverSinks adds to sinks every function that hands one of its own
// string parameters to a sink as the SQL, repeating until nothing
// changes, and returns, per such function, the parameter that carries it:
// a call inside it on that parameter is not a site, its callers' are.
func discoverSinks(pkgs map[string][]file, sinks map[string]sink) map[*ast.FuncDecl]string {
	through := map[*ast.FuncDecl]string{}
	for changed := true; changed; {
		changed = false
		for _, files := range pkgs {
			for _, f := range files {
				for _, d := range f.ast.Decls {
					fn, ok := d.(*ast.FuncDecl)
					if !ok || fn.Body == nil {
						continue
					}
					if _, done := through[fn]; done {
						continue
					}
					names, isString, variadic := paramList(fn)
					ast.Inspect(fn.Body, func(n ast.Node) bool {
						c, ok := n.(*ast.CallExpr)
						if !ok {
							return true
						}
						s, ok := sinks[calleeName(c)]
						if !ok || len(c.Args) <= s.sql {
							return true
						}
						id, ok := c.Args[s.sql].(*ast.Ident)
						if !ok {
							return true
						}
						for p, name := range names {
							if name != id.Name || !isString[p] {
								continue
							}
							args := -1
							if variadic && s.args >= 0 && c.Ellipsis.IsValid() {
								if last, ok := c.Args[len(c.Args)-1].(*ast.Ident); ok && last.Name == names[len(names)-1] {
									args = len(names) - 1
								}
							}
							if _, known := sinks[fn.Name.Name]; !known {
								sinks[fn.Name.Name] = sink{sql: p, args: args}
							}
							through[fn] = name
							changed = true
							return false
						}
						return true
					})
				}
			}
		}
	}
	return through
}

// packageDecls collects what a package declares at its top level that a
// SQL argument may name: constants, variables that nothing in the package
// assigns again, and struct types.
func packageDecls(files []file) (map[string]ast.Expr, map[string]*ast.StructType) {
	globals := map[string]ast.Expr{}
	vars := map[string]ast.Expr{}
	types := map[string]*ast.StructType{}
	assigned := map[string]bool{}
	for _, f := range files {
		ast.Inspect(f.ast, func(n ast.Node) bool {
			if a, ok := n.(*ast.AssignStmt); ok && a.Tok != token.DEFINE {
				for _, l := range a.Lhs {
					if id, ok := l.(*ast.Ident); ok {
						assigned[id.Name] = true
					}
				}
			}
			return true
		})
		for _, d := range f.ast.Decls {
			g, ok := d.(*ast.GenDecl)
			if !ok {
				continue
			}
			for _, s := range g.Specs {
				switch s := s.(type) {
				case *ast.ValueSpec:
					into := globals
					if g.Tok == token.VAR {
						into = vars
					}
					for i, n := range s.Names {
						if i < len(s.Values) {
							into[n.Name] = s.Values[i]
						}
					}
				case *ast.TypeSpec:
					if st, ok := s.Type.(*ast.StructType); ok {
						types[s.Name.Name] = st
					}
				}
			}
		}
	}
	for name, v := range vars {
		if !assigned[name] {
			globals[name] = v
		}
	}
	return globals, types
}

// funcSites returns the SQL sites in one function.
func funcSites(fset *token.FileSet, rel string, fn *ast.FuncDecl, sinks map[string]sink, through string, consts map[string]ast.Expr, types map[string]*ast.StructType) []site {
	r := newResolver(fn, consts, types)
	key := rel + ":" + fn.Name.Name

	// Locals bound to a prepared statement or to a query's rows, and the
	// calls they came from.
	stmtOf := map[string]*ast.CallExpr{}
	rowsOf := map[string]*ast.CallExpr{}
	ast.Inspect(fn.Body, func(n ast.Node) bool {
		a, ok := n.(*ast.AssignStmt)
		if !ok || len(a.Rhs) != 1 || len(a.Lhs) == 0 {
			return true
		}
		c, ok := a.Rhs[0].(*ast.CallExpr)
		id, isID := a.Lhs[0].(*ast.Ident)
		if !ok || !isID {
			return true
		}
		switch calleeName(c) {
		case "PrepareContext":
			stmtOf[id.Name] = c
		case "QueryContext":
			rowsOf[id.Name] = c
		}
		return true
	})

	byCall := map[*ast.CallExpr]*site{}
	var order []*ast.CallExpr
	ast.Inspect(fn.Body, func(n ast.Node) bool {
		c, ok := n.(*ast.CallExpr)
		if !ok {
			return true
		}
		name := calleeName(c)
		if sel, ok := c.Fun.(*ast.SelectorExpr); ok {
			if x, ok := sel.X.(*ast.Ident); ok {
				if prep, ok := stmtOf[x.Name]; ok && stmtCalls[name] {
					if s := byCall[prep]; s != nil && !c.Ellipsis.IsValid() {
						s.nargs = append(s.nargs, len(c.Args)-1)
					}
					return true
				}
				if q, ok := rowsOf[x.Name]; ok && name == "Scan" {
					if s := byCall[q]; s != nil && !c.Ellipsis.IsValid() {
						s.scans = append(s.scans, len(c.Args))
					}
					return true
				}
			}
		}
		sk, ok := sinks[name]
		if !ok || len(c.Args) <= sk.sql {
			return true
		}
		if id, ok := c.Args[sk.sql].(*ast.Ident); ok && through != "" && id.Name == through {
			return true // passes its caller's SQL on; the callers are the sites
		}
		s := &site{pos: fmt.Sprintf("%s:%d", rel, fset.Position(c.Pos()).Line), key: key}
		s.sqls, s.why = r.resolve(c.Args[sk.sql])
		if sk.args >= 0 && !c.Ellipsis.IsValid() {
			s.nargs = append(s.nargs, len(c.Args)-sk.args)
		}
		_, base := baseSinks[name]
		s.read = base && readers[name]
		byCall[c] = s
		order = append(order, c)
		return true
	})
	// A chained x.QueryRowContext(…).Scan(…) is the parent of its query in
	// the tree, so it is visited first: match those once every site is known.
	ast.Inspect(fn.Body, func(n ast.Node) bool {
		c, ok := n.(*ast.CallExpr)
		if !ok || calleeName(c) != "Scan" || c.Ellipsis.IsValid() {
			return true
		}
		if sel, ok := c.Fun.(*ast.SelectorExpr); ok {
			if q, ok := sel.X.(*ast.CallExpr); ok {
				if s := byCall[q]; s != nil {
					s.scans = append(s.scans, len(c.Args))
				}
			}
		}
		return true
	})
	out := make([]site, 0, len(order))
	for _, c := range order {
		out = append(out, *byCall[c])
	}
	return out
}

import (

// memDB is an in-memory database with one table, t(a INTEGER, b TEXT).
func memDB(t *testing.T) *sql.DB {
	t.Helper()
	db, err := sql.Open("sqlite", ":memory:")
	if err != nil {
		t.Fatal(err)
	}
	db.SetMaxOpenConns(1) // one connection: an in-memory database is per connection
	t.Cleanup(func() { db.Close() })
	if _, err := db.Exec(`CREATE TABLE t (a INTEGER, b TEXT)`); err != nil {
		t.Fatal(err)
	}
	return db
}

func TestProblems(t *testing.T) {
	db := memDB(t)
	cases := map[string]struct {
		site site
		want string // a substring of the one problem expected; "" for none
	}{
		"clean":            {site{sqls: []string{"SELECT a, b FROM t WHERE a = ?"}, nargs: []int{1}, read: true, scans: []int{2}}, ""},
		"unknown column":   {site{sqls: []string{"SELECT c FROM t"}}, "no such column"},
		"unknown table":    {site{sqls: []string{"DELETE FROM u"}}, "no such table"},
		"second statement": {site{sqls: []string{"SELECT a FROM t; SELECT z FROM t"}}, "no such column"},
		"too few args":     {site{sqls: []string{"UPDATE t SET b = ? WHERE a = ?"}, nargs: []int{1}}, "passes 1 arguments; the statement binds 2"},
		"too many args":    {site{sqls: []string{"SELECT a FROM t WHERE a = ?1 OR b = ?1"}, nargs: []int{2}}, "passes 2 arguments; the statement binds 1"},
		"scan short":       {site{sqls: []string{"SELECT a, b FROM t"}, read: true, scans: []int{1}}, "scans 1 values; the statement returns 2 columns"},
		"limit param":      {site{sqls: []string{"SELECT a FROM t LIMIT ?"}, nargs: []int{1}, read: true, scans: []int{1}}, ""},
		"named params":     {site{sqls: []string{"SELECT a FROM t WHERE a = :n OR a = :n LIMIT :m"}, nargs: []int{2}, read: true, scans: []int{1}}, ""},
		"no statement":     {site{sqls: []string{"-- nothing"}}, "no statement"},
	}
	for name, c := range cases {

		switch {
		case c.want == "" && len(got) != 0:
			t.Errorf("%s: got %q, want none", name, got)

func TestColumnsRollsBack(t *testing.T) {

	if _, err := columns(context.Background(), db, "SELECT a FROM t", p); err != nil {
		t.Fatal(err)
	}
	if _, err := db.Exec(`INSERT INTO t VALUES (1, 'x')`); err != nil {
		t.Fatal(err)

package sqlcheck

import (
	"context"
	"database/sql"
	"fmt"
	"strconv"
)

// problems checks one site's SQL against db: every statement prepares
// (D4), a lone statement binds as many parameters as each call passes
// (D5), and a read's rows have as many columns as each Scan reads (D6).
// It returns nothing for a dynamic site: TestDynamicSitesAreListed owns those.
func problems(ctx context.Context, db *sql.DB, s site) []string {
	var out []string
	for _, q := range s.sqls {
		stmts, err := statements(q)
		if err != nil {
			out = append(out, err.Error())
			continue
		}
		if len(stmts) == 0 {
			out = append(out, "no statement in "+strconv.Quote(q))
			continue
		}
		prepared := true
		for _, st := range stmts {
			stmt, err := db.PrepareContext(ctx, st)
			if err != nil {
				out = append(out, fmt.Sprintf("%v\n%s", err, st))
				prepared = false
				continue
			}
			stmt.Close()
		}
		// Counts are per statement: what a script binds across its
		// statements is the driver's business, and no Go SQL string is one.
		if !prepared || len(stmts) != 1 {
			continue
		}
		st := stmts[0]
		p, err := bindings(st)
		if err != nil {
			out = append(out, err.Error())
			continue
		}
		for _, n := range s.nargs {
			if n != p.max {
				out = append(out, fmt.Sprintf("passes %d arguments; the statement binds %d\n%s", n, p.max, st))
			}
		}
		if !s.read || len(s.scans) == 0 {
			continue
		}
		cols, err := columns(ctx, db, st, p)
		if err != nil {
			out = append(out, fmt.Sprintf("running it to count its columns: %v\n%s", err, st))
			continue
		}
		for _, n := range s.scans {
			if n != cols {
				out = append(out, fmt.Sprintf("scans %d values; the statement returns %d columns\n%s", n, cols, st))
			}
		}
	}
	return out
}

// columns runs a read once on db, inside a transaction it rolls back,
// and returns how many columns its rows have. Every parameter is bound to
// NULL, then to 0 if that fails (LIMIT NULL is a datatype mismatch).
func columns(ctx context.Context, db *sql.DB, stmt string, p params) (int, error) {
	try := func(zero any) (int, error) {
		tx, err := db.BeginTx(ctx, nil)
		if err != nil {
			return 0, err
		}
		defer tx.Rollback()
		args := make([]any, 0, p.max)
		for i := 1; i <= p.max; i++ {
			if name, ok := p.names[i]; ok {
				args = append(args, sql.Named(name, zero))
			} else {
				args = append(args, zero)
			}
		}
		rows, err := tx.QueryContext(ctx, stmt, args...)
		if err != nil {
			return 0, err
		}
		defer rows.Close()
		cols, err := rows.Columns()
		if err != nil {
			return 0, err
		}
		return len(cols), rows.Err()
	}
	n, err := try(nil)
	if err != nil {
		n, err = try(0)
	}
	return n, err
}
